export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({error:"Method not allowed"}); } const key = process.env.OPENAI_API_KEY; if (!key) { return res.status(500).json({error:"AI key is not configured"}); } const messages = req.body?.messages; const image = req.body?.image; if (!Array.isArray(messages)) { return res.status(400).json({error:"Invalid messages"}); } const input = messages.slice(-12).map((m, index) => { const content = [{type:"input_text",text:String(m.content).slice(0,4000)}]; if (index === messages.slice(-12).length - 1 && image && /^data:image\/(jpeg|png|webp);base64,/.test(image)) { content.push({type:"input_image",image_url:image}); } return { role: m.role === "assistant" ? "assistant" : "user", content }; }); try { const response = await fetch("https://api.openai.com/v1/responses", { method:"POST", headers:{ "Content-Type":"application/json", "Authorization":"Bearer " + key }, body:JSON.stringify({ model:"gpt-4.1-mini", instructions:"You are a helpful AI assistant. You can answer questions and analyze uploaded images.", input, max_output_tokens:1200 }) }); const data = await response.json(); if (!response.ok) { return res.status(502).json({error:"AI service error"}); } const reply = (data.output || []) .filter(item => item.type === "message") .flatMap(item => item.content || []) .filter(item => item.type === "output_text") .map(item => item.text) .join("\n"); return res.status(200).json({ reply: reply || "I couldn't generate an answer." }); } catch (error) { return res.status(500).json({error:"Could not connect to AI"}); } }
